from ayx import Alteryx

import numpy as np
import pandas as pd

def get_coords(z):
    a = z//144
    b = z%144
    
    y = (a+1/2)*180/91 - 90
    x = (b+1/2)*360/144 - 180

    return x,y


r = np.array([i for i in range(13104)])
coords = []

for z in r:
    x,y = get_coords(z)
    coords.append([x,y])

coords = pd.DataFrame(coords, columns = ['longitude', 'latitude'])
